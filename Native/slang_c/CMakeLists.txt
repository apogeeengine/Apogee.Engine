# The C API over Slang (slang.h is C++ only). It links the vendored Slang compiler library, which has to sit next to it at run
# time: ThirdParty/NativeLibraries.props copies both into every output folder, and the rpath is relative to this library.
set(SLANG "${APOGEE_THIRDPARTY}/slang")
set(SLANG_LIBS "${SLANG}/lib/${APOGEE_OS}-${APOGEE_ARCH}")
if(WIN32)
  set(SLANG_COMPILER "${SLANG_LIBS}/slang-compiler.lib")
elseif(APPLE)
  set(SLANG_COMPILER "${SLANG_LIBS}/libslang-compiler.0.2026.19.dylib")
else()
  set(SLANG_COMPILER "${SLANG_LIBS}/libslang-compiler.so.0.2026.19")
endif()

add_library(slang_c SHARED apogee_slang.cpp)
target_include_directories(slang_c PRIVATE "${SLANG}/include")
target_link_libraries(slang_c PRIVATE "${SLANG_COMPILER}")
if(WIN32)
  target_compile_definitions(slang_c PRIVATE "APOGEE_SLANG_API=__declspec(dllexport)")
else()
  target_compile_definitions(slang_c PRIVATE "APOGEE_SLANG_API=__attribute__((visibility(\"default\")))")
  set_target_properties(slang_c PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$<IF:$<BOOL:${APPLE}>,@loader_path,$ORIGIN>")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(slang_c PRIVATE -Wl,--no-undefined)
endif()
apogee_library(slang_c)
